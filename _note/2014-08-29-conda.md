if [ $? -eq 0 ]; then
    eval "$__conda_setup"
else
    if [ -f "/home/aaron/miniforge3/etc/profile.d/conda.sh" ]; then
        . "/home/aaron/miniforge3/etc/profile.d/conda.sh"
    else
        export PATH="/home/aaron/miniforge3/bin:$PATH"
    fi
fi
unset __conda_setup
